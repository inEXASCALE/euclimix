cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 80)
endif()
project(euclimix LANGUAGES CXX CUDA)
find_package(Torch REQUIRED)
find_package(CUDAToolkit 12.8 REQUIRED)
# Follow the standard required by the installed LibTorch (17 for the paper
# environment, 20 for newer builds), for both host and device translation units.
get_target_property(EUCLIMIX_CXX_STANDARD torch CXX_STANDARD)
if(NOT EUCLIMIX_CXX_STANDARD)
  get_target_property(EUCLIMIX_TORCH_CXX_FEATURES torch INTERFACE_COMPILE_FEATURES)
  if("cxx_std_20" IN_LIST EUCLIMIX_TORCH_CXX_FEATURES)
    set(EUCLIMIX_CXX_STANDARD 20)
  else()
    set(EUCLIMIX_CXX_STANDARD 17)
  endif()
endif()
add_library(euclimix SHARED paper.cu controlled.cu)
target_include_directories(euclimix PUBLIC include)
target_link_libraries(euclimix PUBLIC ${TORCH_LIBRARIES} CUDA::cublas CUDA::cudart)
target_compile_features(euclimix PUBLIC cxx_std_${EUCLIMIX_CXX_STANDARD})
set_property(TARGET euclimix PROPERTY CUDA_STANDARD ${EUCLIMIX_CXX_STANDARD})
set_property(TARGET euclimix PROPERTY CUDA_STANDARD_REQUIRED ON)
set_source_files_properties(paper.cu PROPERTIES COMPILE_OPTIONS "-O3;--use_fast_math")
set_source_files_properties(controlled.cu PROPERTIES COMPILE_OPTIONS "-O3;--ftz=false;--fmad=false;--prec-div=true;--prec-sqrt=true")
separate_arguments(TORCH_CXX_FLAGS_LIST NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(euclimix PUBLIC ${TORCH_CXX_FLAGS_LIST})
option(EUCLIMIX_BUILD_EXAMPLE "Build a standalone native smoke test" OFF)
if(EUCLIMIX_BUILD_EXAMPLE)
  add_executable(euclimix_native_example native_example.cpp)
  target_link_libraries(euclimix_native_example PRIVATE euclimix)
  add_executable(euclimix_state_example state_example.cpp)
  target_link_libraries(euclimix_state_example PRIVATE euclimix)
endif()
install(TARGETS euclimix LIBRARY DESTINATION lib)
install(DIRECTORY include/euclimix DESTINATION include)
